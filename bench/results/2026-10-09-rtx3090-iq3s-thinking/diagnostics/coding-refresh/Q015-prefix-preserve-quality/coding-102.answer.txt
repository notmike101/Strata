def topological_order(nodes, edges):
    valid_nodes = set(nodes)
    adjacency = {node: set() for node in valid_nodes}
    indegree = {node: 0 for node in valid_nodes}

    for before, after in edges:
        if before not in valid_nodes or after not in valid_nodes:
            raise ValueError("edge endpoint not in nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def sift_down(heap, start):
        pos = start
        end = len(heap)

        while True:
            left = 2 * pos + 1
            right = left + 1
            target = pos

            if left < end and heap[left] < heap[target]:
                target = left
            if right < end and heap[right] < heap[target]:
                target = right

            if target == pos:
                return

            heap[pos], heap[target] = heap[target], heap[pos]
            pos = target

    def sift_up(heap, start):
        pos = start

        while pos > 0:
            parent = (pos - 1) // 2

            if heap[pos] < heap[parent]:
                heap[pos], heap[parent] = heap[parent], heap[pos]
                pos = parent
            else:
                return

    heap = [node for node in valid_nodes if indegree[node] == 0]

    for i in range(len(heap) // 2 - 1, -1, -1):
        sift_down(heap, i)

    order = []

    while heap:
        node = heap[0]
        last = heap.pop()

        if heap:
            heap[0] = last
            sift_down(heap, 0)

        order.append(node)

        for successor in adjacency[node]:
            indegree[successor] -= 1

            if indegree[successor] == 0:
                heap.append(successor)
                sift_up(heap, len(heap) - 1)

    if len(order) != len(valid_nodes):
        raise ValueError("cycle detected")

    return order
