def topological_order(nodes, edges):
    def _sift_down(heap, i):
        n = len(heap)
        item = heap[i]
        while True:
            left = 2 * i + 1
            if left >= n:
                break
            right = left + 1
            child = left
            if right < n and heap[right] < heap[left]:
                child = right
            if heap[child] >= item:
                break
            heap[i] = heap[child]
            i = child
        heap[i] = item

    def _heappush(heap, item):
        heap.append(item)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= item:
                break
            heap[i] = heap[parent]
            i = parent
        heap[i] = item

    def _heappop(heap):
        root = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            _sift_down(heap, 0)
        return root

    node_set = set(nodes)

    indegree = {node: 0 for node in node_set}
    successors = {node: [] for node in node_set}
    seen_edges = set()

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")

        edge = (before, after)
        if edge in seen_edges:
            continue

        seen_edges.add(edge)
        successors[before].append(after)
        indegree[after] += 1

    available = []
    for node in node_set:
        if indegree[node] == 0:
            _heappush(available, node)

    order = []
    while available:
        node = _heappop(available)
        order.append(node)

        for nxt in successors[node]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                _heappush(available, nxt)

    if len(order) != len(node_set):
        raise ValueError("cycle detected")

    return order
