def topological_order(nodes, edges):
    nodes_set = set(nodes)
    adjacency = {node: set() for node in nodes_set}
    indegree = {node: 0 for node in nodes_set}

    for before, after in edges:
        if before not in nodes_set or after not in nodes_set:
            raise ValueError("edge endpoint absent from nodes")
        if before == after:
            raise ValueError("cycle detected")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = {node for node in nodes_set if indegree[node] == 0}
    order = []
    while available:
        node = min(available)
        available.remove(node)
        order.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                available.add(neighbor)
    if len(order) != len(nodes_set):
        raise ValueError("cycle detected")
    return order

```python
def topological_order(nodes, edges):
    nodes_set = set(nodes)
    adjacency = {node: set() for node in nodes_set}
    indegree = {node: 0 for node in nodes_set}

    for before, after in edges:
        if before not in nodes_set or after not in nodes_set:
            raise ValueError("edge endpoint absent from nodes")

        if before == after:
            raise ValueError("cycle detected")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = {node for node in nodes_set if indegree[node] == 0}
    order = []

    while available:
        node = min(available)
        available.remove(node)
        order.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                available.add(neighbor)

    if len(order) != len(nodes_set):
        raise ValueError("cycle detected")

    return order
```
