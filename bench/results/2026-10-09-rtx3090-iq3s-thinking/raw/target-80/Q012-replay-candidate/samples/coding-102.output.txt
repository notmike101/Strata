def topological_order(nodes, edges):
    node_set = set(nodes)
    indegree = {}
    adjacency = {}

    for node in nodes:
        indegree[node] = 0
        adjacency[node] = set()

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("Edge endpoint not found in nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    heap = [node for node in nodes if indegree[node] == 0]

    def sift_down(index):
        node = heap[index]
        child = 2 * index + 1
        while child < len(heap):
            right = child + 1
            if right < len(heap) and heap[right] < heap[child]:
                child = right
            if heap[child] >= node:
                break
            heap[index] = heap[child]
            index = child
            child = 2 * index + 1
        heap[index] = node

    def sift_up(index):
        node = heap[index]
        while index > 0:
            parent = (index - 1) // 2
            if heap[parent] <= node:
                break
            heap[index] = heap[parent]
            index = parent
        heap[index] = node

    for i in range(len(heap) // 2 - 1, -1, -1):
        sift_down(i)

    order = []
    while heap:
        node = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            sift_down(0)
        order.append(node)

        for nxt in adjacency[node]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                heap.append(nxt)
                sift_up(len(heap) - 1)

    if len(order) != len(node_set):
        raise ValueError("Graph contains a cycle")

    return order

    for node in nodes:
        indegree[node] = 0
        adjacency[node] = set()

                break

            index = child
            child = 2 * index + 1
        heap[index] = node

        while index > 0:

            if heap[parent] <= node:
                break
            heap[index] = heap[parent]
