def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    def _sift_down(heap, index, size):
        while True:
            left = 2 * index + 1
            right = left + 1
            smallest = index
            if left < size and heap[left] < heap[smallest]:
                smallest = left
            if right < size and heap[right] < heap[smallest]:
                smallest = right
            if smallest == index:
                return
            heap[index], heap[smallest] = heap[smallest], heap[index]
            index = smallest

    def _push(heap, item):
        heap.append(item)
        index = len(heap) - 1
        while index > 0:
            parent = (index - 1) // 2
            if heap[parent] <= heap[index]:
                return
            heap[parent], heap[index] = heap[index], heap[parent]
            index = parent

    def _pop(heap):
        if not heap:
            return None
        smallest = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            _sift_down(heap, 0, len(heap))
        return smallest

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("edges must contain pairs") from None
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")
        if after not in adjacency[before]:

            indegree[after] += 1

        ordering.append(node)

            indegree[neighbor] -= 1

def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    def _sift_down(heap, index, size):
        while True:
            left = 2 * index + 1
            right = left + 1
            smallest = index
            if left < size and heap[left] < heap[smallest]:
                smallest = left
            if right < size and heap[right] < heap[smallest]:
                smallest = right
            if smallest == index:
                return
            heap[index], heap[smallest] = heap[smallest], heap[index]
            index = smallest

    def _push(heap, item):
        heap.append(item)
        index = len(heap) - 1
        while index > 0:
            parent = (index - 1) // 2
            if heap[parent] <= heap[index]:
                return
            heap[parent], heap[index] = heap[index], heap[parent]
            index = parent

    def _pop(heap):
        if not heap:
            return None
        smallest = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            _sift_down(heap, 0, len(heap))
        return smallest

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("edges must contain pairs")

        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = sorted(node for node in node_set if indegree[node] == 0)
    ordering = []

    while available:
        node = _pop(available)
        ordering.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                _push(available, neighbor)

    if len(ordering) != len(node_set):
        raise ValueError("cycle detected")

    return ordering

            heap[index], heap[smallest] = heap[smallest], heap[index]
            index = smallest

        heap.append(item)
        index = len(heap) - 1
        while index > 0:
            parent = (index - 1) // 2
            if heap[parent] <= heap[index]:
                return
            heap[parent], heap[index] = heap[index], heap[parent]
            index = parent

    def _pop(heap):
        if not heap:
            return None
        smallest = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            _sift_down(heap, 0, len(heap))
        return smallest

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("edges must contain pairs")

        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = sorted(node for node in node_set if indegree[node] == 0)
    ordering = []

    while available:
        node = _pop(available)
        ordering.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                _push(available, neighbor)

    if len(ordering) != len(node_set):
        raise ValueError("cycle detected")

    return ordering

```python
def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    def _sift_down(heap, index, size):
        while True:
            left = 2 * index + 1
            right = left + 1
            smallest = index

            if left < size and heap[left] < heap[smallest]:
                smallest = left
            if right < size and heap[right] < heap[smallest]:
                smallest = right

            if smallest == index:
                return

            heap[index], heap[smallest] = heap[smallest], heap[index]
            index = smallest

    def _push(heap, item):
        heap.append(item)
        index = len(heap) - 1

        while index > 0:
            parent = (index - 1) // 2
            if heap[parent] <= heap[index]:
                return

            heap[parent], heap[index] = heap[index], heap[parent]
            index = parent

    def _pop(heap):
        if not heap:
            return None

        smallest = heap[0]
        last = heap.pop()

        if heap:
            heap[0] = last
            _sift_down(heap, 0, len(heap))

        return smallest

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("edges must contain pairs")

        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = sorted(node for node in node_set if indegree[node] == 0)
    ordering = []

    while available:
        node = _pop(available)
        ordering.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                _push(available, neighbor)

    if len(ordering) != len(node_set):
        raise ValueError("cycle detected")

    return ordering
