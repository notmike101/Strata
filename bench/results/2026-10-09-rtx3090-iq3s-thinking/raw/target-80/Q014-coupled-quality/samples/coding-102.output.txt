        while index > 0:
            parent = (index - 1) // 2
            if heap[parent] <= heap[index]:
                break
            heap[parent], heap[index] = heap[index], heap[parent]
            index = parent

    node_set = set(nodes)
    node_list = list(node_set)

    indegree = {node: 0 for node in node_list}
    adjacency = {node: set() for node in node_list}

    for edge in edges:
        before, after = edge

        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    heap = [node for node in node_list if indegree[node] == 0]

    for i in range(len(heap) // 2 - 1, -1, -1):
        sift_down(heap, i)

    order = []

    while heap:
        if len(heap) == 1:
            node = heap.pop()
        else:
            node = heap[0]
            heap[0] = heap.pop()
            sift_down(heap, 0)

        order.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                heap.append(neighbor)
                sift_up(heap, len(heap) - 1)

    if len(order) != len(node_list):
        raise ValueError("cycle detected")

    return order
