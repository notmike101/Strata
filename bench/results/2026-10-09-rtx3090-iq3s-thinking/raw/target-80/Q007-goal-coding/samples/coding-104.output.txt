def topological_order(nodes, edges):
    node_set = set(nodes)

    def _in_nodes(value):
        try:
            return value in node_set
        except TypeError:
            return False

    adjacency = {node: set() for node in nodes}
    indegree = {node: 0 for node in nodes}

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("each edge must contain exactly two endpoints")

        if not _in_nodes(before) or not _in_nodes(after):
            raise ValueError("edge endpoint absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = [node for node in node_set if indegree[node] == 0]

    def _heapify(heap):
        for i in range(len(heap) // 2 - 1, -1, -1):
            while True:
                left = 2 * i + 1
                right = left + 1
                smallest = i
                if left < len(heap) and heap[left] < heap[smallest]:
                    smallest = left
                if right < len(heap) and heap[right] < heap[smallest]:
                    smallest = right
                if smallest == i:
                    break
                heap[i], heap[smallest] = heap[smallest], heap[i]
                i = smallest

    def _heappush(heap, value):
        heap.append(value)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    def _heappop(heap):
        if not heap:
            return None
        root = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            i = 0
            while True:
                left = 2 * i + 1
                right = left + 1
                smallest = i
                if left < len(heap) and heap[left] < heap[smallest]:
                    smallest = left
                if right < len(heap) and heap[right] < heap[smallest]:
                    smallest = right
                if smallest == i:
                    break
                heap[i], heap[smallest] = heap[smallest], heap[i]
                i = smallest
        return root

    _heapify(available)
    order = []

    while available:
        current = _heappop(available)
        order.append(current)
        for nxt in adjacency[current]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                _heappush(available, nxt)

    if len(order) != len(node_set):
        raise ValueError("graph contains a cycle")

def topological_order(nodes, edges):
    node_set = set(nodes)

    def _in_nodes(value):
        try:
            return value in node_set
        except TypeError:
            return False

    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("each edge must contain exactly two endpoints")

        if not _in_nodes(before) or not _in_nodes(after):
            raise ValueError("edge endpoint absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = [node for node in node_set if indegree[node] == 0]

    def _heapify(heap):
        for i in range(len(heap) // 2 - 1, -1, -1):
            while True:
                left = 2 * i + 1
                right = left + 1
                smallest = i

                if left < len(heap) and heap[left] < heap[smallest]:
                    smallest = left
                if right < len(heap) and heap[right] < heap[smallest]:
                    smallest = right
                if smallest == i:
                    break

                heap[i], heap[smallest] = heap[smallest], heap[i]
                i = smallest

    def _heappush(heap, value):
        heap.append(value)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    def _heappop(heap):
        if not heap:
            return None

        root = heap[0]
        last = heap.pop()

        if heap:
            heap[0] = last
            i = 0
            while True:
                left = 2 * i + 1
                right = left + 1
                smallest = i

                if left < len(heap) and heap[left] < heap[smallest]:
                    smallest = left
                if right < len(heap) and heap[right] < heap[smallest]:
                    smallest = right
                if smallest == i:
                    break

                heap[i], heap[smallest] = heap[smallest], heap[i]
                i = smallest

        return root

    _heapify(available)
    order = []

    while available:
        current = _heappop(available)
        order.append(current)

        for nxt in adjacency[current]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                _heappush(available, nxt)

    if len(order) != len(node_set):
        raise ValueError("graph contains a cycle")

    return order
