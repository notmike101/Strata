    node_set = set(nodes)
    adj = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for before, after in edges:
        if before not in node_set or after not in node_set:

        if after not in adj[before]:
            adj[before].add(after)
            indegree[after] += 1

    def _sift_up(h, i):
        while i > 0:
            p = (i - 1) // 2
            if h[i] < h[p]:
                h[i], h[p] = h[p], h[i]
                i = p
            else:
                break

    def _sift_down(h, i):
        n = len(h)
        while True:
            l = 2*i + 1
            r = l + 1
            smallest = i
            if l < n and h[l] < h[smallest]: smallest = l
            if r < n and h[r] < h[smallest]: smallest = r
            if smallest == i: break
            h[i], h[smallest] = h[smallest], h[i]
            i = smallest

    def _heapify(h):
        for i in range(len(h)//2 - 1, -1, -1):
            _sift_down(h, i)

    def _heappush(h, x):
        h.append(x)
        _sift_up(h, len(h)-1)

    def _heappop(h):
        if len(h) == 1:
            return h.pop()
        root = h[0]
        last = h.pop()
        h[0] = last
        _sift_down(h, 0)
        return root

    heap = [n for n in node_set if indegree[n] == 0]
    _heapify(heap)

    order = []
    while heap:
        node = _heappop(heap)
        order.append(node)
        for nxt in adj[node]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                _heappush(heap, nxt)

    if len(order) != len(node_set):
        raise ValueError("cycle detected")
    return order

def topological_order(nodes, edges):
    node_set = set(nodes)

    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("Edge endpoint is absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def sift_up(heap, index):
        while index > 0:
            parent = (index - 1) // 2
            if heap[index] < heap[parent]:
                heap[index], heap[parent] = heap[parent], heap[index]
                index = parent
            else:
                break

    def sift_down(heap, index):
        size = len(heap)

        while True:
            left = 2 * index + 1
            right = left + 1
            smallest = index

            if left < size and heap[left] < heap[smallest]:
                smallest = left
            if right < size and heap[right] < heap[smallest]:
                smallest = right

            if smallest == index:
                break

            heap[index], heap[smallest] = heap[smallest], heap[index]
            index = smallest

    def heapify(heap):
        for index in range(len(heap) // 2 - 1, -1, -1):
            sift_down(heap, index)

    def heappush(heap, value):
        heap.append(value)
        sift_up(heap, len(heap) - 1)

    def heappop(heap):
        if len(heap) == 1:
            return heap.pop()

        smallest = heap[0]
        last = heap.pop()
        heap[0] = last
        sift_down(heap, 0)
        return smallest

    heap = [node for node in node_set if indegree[node] == 0]
    heapify(heap)

    order = []

    while heap:
        node = heappop(heap)
        order.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                heappush(heap, neighbor)

    if len(order) != len(node_set):
        raise ValueError("Cycle detected")

    return order
