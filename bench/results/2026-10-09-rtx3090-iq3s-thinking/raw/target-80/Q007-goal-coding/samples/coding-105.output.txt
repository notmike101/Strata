```python
def topological_order(nodes, edges):
    unique_nodes = []
    seen_nodes = set()
    for node in nodes:
        if node not in seen_nodes:
            seen_nodes.add(node)
            unique_nodes.append(node)

    adjacency = {node: set() for node in unique_nodes}
    indegree = {node: 0 for node in unique_nodes}

    for edge in edges:
        if isinstance(edge, (str, bytes, bytearray)):
            raise ValueError("Each edge must be a pair of endpoints")
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("Each edge must be a pair of endpoints")

        try:
            if before not in seen_nodes or after not in seen_nodes:
                raise ValueError("Edge endpoint not found in nodes")
        except TypeError:
            raise ValueError("Edge endpoints must be hashable and present in nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def _push(heap, item):
        heap.append(item)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= item:
                break
            heap[i] = heap[parent]
            i = parent
        heap[i] = item

    def _pop(heap):
        root = heap[0]
        last = heap.pop()
        if heap:
            i = 0
            n = len(heap)
            while True:
                left = 2 * i + 1
                if left >= n:
                    break
                right = left + 1
                child = left
                if right < n and heap[right] < heap[left]:
                    child = right
                if last <= heap[child]:
                    break
                heap[i] = heap[child]
                i = child
            heap[i] = last
        return root

    heap = []
    for node in unique_nodes:
        if indegree[node] == 0:
            _push(heap, node)

    order = []
    while heap:
        current = _pop(heap)
        order.append(current)
        for neighbor in adjacency[current]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                _push(heap, neighbor)

    if len(order) != len(unique_nodes):
        raise ValueError("Cycle detected")

    return order
```
