def topological_order(nodes, edges):
    node_set = set(nodes)

    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("Edge endpoint is absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def sift_up(heap, index):
        while index > 0:
            parent = (index - 1) // 2
            if heap[index] < heap[parent]:
                heap[index], heap[parent] = heap[parent], heap[index]
                index = parent
            else:
                break

    def sift_down(heap, index):
        size = len(heap)

        while True:
            left = 2 * index + 1
            right = left + 1
            smallest = index

            if left < size and heap[left] < heap[smallest]:
                smallest = left
            if right < size and heap[right] < heap[smallest]:
                smallest = right

            if smallest == index:
                break

            heap[index], heap[smallest] = heap[smallest], heap[index]
            index = smallest

    def heapify(heap):
        for index in range(len(heap) // 2 - 1, -1, -1):
            sift_down(heap, index)

    def heappush(heap, value):
        heap.append(value)
        sift_up(heap, len(heap) - 1)

    def heappop(heap):
        if len(heap) == 1:
            return heap.pop()

        smallest = heap[0]
        last = heap.pop()
        heap[0] = last
        sift_down(heap, 0)
        return smallest

    heap = [node for node in node_set if indegree[node] == 0]
    heapify(heap)

    order = []

    while heap:
        node = heappop(heap)
        order.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                heappush(heap, neighbor)

    if len(order) != len(node_set):
        raise ValueError("Cycle detected")

    return order
