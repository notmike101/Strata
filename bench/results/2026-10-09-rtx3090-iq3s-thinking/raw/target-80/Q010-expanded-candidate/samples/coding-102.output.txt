def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in nodes}
    indegree = {node: 0 for node in nodes}

    for edge in edges:
        before, after = edge
        if before not in node_set or after not in node_set:
            raise ValueError("Edge endpoint not found in nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def _push(heap, item):
        heap.append(item)
        i = len(heap) - 1
        while i:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    def _pop(heap):
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            i = 0
            n = len(heap)
            while True:
                left = 2 * i + 1
                if left >= n:
                    break
                right = left + 1
                child = left
                if right < n and heap[right] < heap[left]:
                    child = right
                if heap[i] <= heap[child]:
                    break
                heap[i], heap[child] = heap[child], heap[i]
                i = child
        return top

    heap = []
    for node in nodes:
        if indegree[node] == 0:
            _push(heap, node)

    order = []
    while heap:
        node = _pop(heap)
        order.append(node)
        for nxt in adjacency[node]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                _push(heap, nxt)

    if len(order) != len(node_set):
        raise ValueError("Cycle detected")
    return order

def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in nodes}
    indegree = {node: 0 for node in nodes}

    for edge in edges:
        before, after = edge
        if before not in node_set or after not in node_set:
            raise ValueError("Edge endpoint not found in nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def _push(heap, item):
        heap.append(item)
        i = len(heap) - 1
        while i:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    def _pop(heap):
        top = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            i = 0
            n = len(heap)
            while True:
                left = 2 * i + 1
                if left >= n:
                    break
                right = left + 1
                child = left
                if right < n and heap[right] < heap[left]:
                    child = right
                if heap[i] <= heap[child]:
                    break
                heap[i], heap[child] = heap[child], heap[i]
                i = child
        return top

    heap = []
    for node in node_set:
        if indegree[node] == 0:
            _push(heap, node)

    order = []
    while heap:
        node = _pop(heap)
        order.append(node)
        for nxt in adjacency[node]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                _push(heap, nxt)

    if len(order) != len(node_set):
        raise ValueError("Cycle detected")

    return order
