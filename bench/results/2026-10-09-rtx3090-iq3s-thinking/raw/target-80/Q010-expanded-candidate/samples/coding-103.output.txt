```python
def topological_order(nodes, edges):
    node_set = set(nodes)
    indegree = {node: 0 for node in nodes}
    outgoing = {node: [] for node in nodes}
    seen_edges = set()

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("invalid edge")

        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint not in nodes")

        edge_pair = (before, after)
        if edge_pair in seen_edges:
            continue
        seen_edges.add(edge_pair)

        outgoing[before].append(after)
        indegree[after] += 1

    heap = [node for node in nodes if indegree[node] == 0]

    def sift_down(i):
        size = len(heap)
        while True:
            left = 2 * i + 1
            right = left + 1
            target = i
            if left < size and heap[left] < heap[target]:
                target = left
            if right < size and heap[right] < heap[target]:
                target = right
            if target == i:
                break
            heap[i], heap[target] = heap[target], heap[i]
            i = target

    def sift_up(i):
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    for i in range(len(heap) // 2 - 1, -1, -1):
        sift_down(i)

    order = []
    while heap:
        current = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            sift_down(0)

        order.append(current)

        for nxt in outgoing[current]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                heap.append(nxt)
                sift_up(len(heap) - 1)

    if len(order) != len(node_set):
        raise ValueError("cycle detected")

    return order
```

```python
def topological_order(nodes, edges):
    node_set = set(nodes)
    indegree = {node: 0 for node in node_set}
    outgoing = {node: set() for node in node_set}

    for edge in edges:
        if isinstance(edge, (str, bytes, bytearray)):
            raise ValueError("invalid edge")
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("invalid edge")

        try:
            if before not in node_set or after not in node_set:
                raise ValueError("edge endpoint not in nodes")
        except TypeError:
            raise ValueError("edge endpoint not in nodes")

        if after in outgoing[before]:
            continue
        outgoing[before].add(after)
        indegree[after] += 1

    heap = [node for node in node_set if indegree[node] == 0]

    def sift_down(i):
        size = len(heap)
        while True:
            left = 2 * i + 1
            right = left + 1
            smallest = i
            if left < size and heap[left] < heap[smallest]:
                smallest = left
            if right < size and heap[right] < heap[smallest]:
                smallest = right
            if smallest == i:
                break
            heap[i], heap[smallest] = heap[smallest], heap[i]
            i = smallest

    def sift_up(i):
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    for i in range(len(heap) // 2 - 1, -1, -1):
        sift_down(i)

    order = []
    while heap:
        current = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            sift_down(0)

        order.append(current)

        for nxt in outgoing[current]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                heap.append(nxt)
                sift_up(len(heap) - 1)

    if len(order) != len(node_set):
        raise ValueError("cycle detected")

    return order
```
