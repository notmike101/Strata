```python
def topological_order(nodes, edges):
    node_set = set(nodes)
    indegree = {node: 0 for node in nodes}
    adjacency = {node: set() for node in indegree}

    for edge in edges:
        before, after = edge
        if before not in node_set or after not in node_set:
            raise ValueError("Edge endpoints must be in nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = sorted(node for node, degree in indegree.items() if degree == 0)
    order = []

    while available:
        node = available.pop(0)
        order.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                pos = len(available)
                while pos > 0 and available[pos - 1] > neighbor:
                    pos -= 1
                available.insert(pos, neighbor)

    if len(order) != len(indegree):
        raise ValueError("Input graph contains a cycle")

    return order
```

def topological_order(nodes, edges):
    def _push(heap, value):
        heap.append(value)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= value:
                break
            heap[i] = heap[parent]
            i = parent
        heap[i] = value

    def _pop(heap):
        root = heap[0]
        heap[0] = heap[-1]
        heap.pop()
        if not heap:
            return root

        i = 0
        n = len(heap)
        while True:
            left = 2 * i + 1
            if left >= n:
                break
            right = left + 1
            child = left
            if right < n and heap[right] < heap[left]:
                child = right
            if heap[i] <= heap[child]:
                break
            heap[i], heap[child] = heap[child], heap[i]
            i = child
        return root

    node_set = set(nodes)
    indegree = {node: 0 for node in node_set}
    adjacency = {node: set() for node in node_set}

    for edge in edges:
        before, after = edge
        if before not in node_set or after not in node_set:
            raise ValueError("Edge endpoint absent from nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = []
    for node, degree in indegree.items():
        if degree == 0:
            _push(available, node)

    order = []
    while available:
        node = _pop(available)
        order.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                _push(available, neighbor)

    if len(order) != len(indegree):
        raise ValueError("Cycle detected")
    return order
```

def topological_order(nodes, edges):
    def _push(heap, value):
        heap.append(value)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= value:
                break
            heap[i] = heap[parent]
            i = parent
        heap[i] = value

    def _pop(heap):
        root = heap[0]
        heap[0] = heap[-1]
        heap.pop()

        if not heap:
            return root

        i = 0
        n = len(heap)
        while True:
            left = 2 * i + 1
            if left >= n:
                break

            right = left + 1
            child = left
            if right < n and heap[right] < heap[left]:
                child = right

            if heap[i] <= heap[child]:
                break

            heap[i], heap[child] = heap[child], heap[i]
            i = child

        return root

    node_set = set(nodes)
    indegree = {node: 0 for node in node_set}
    adjacency = {node: set() for node in node_set}

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("Edge endpoint absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = []
    for node, degree in indegree.items():
        if degree == 0:
            _push(available, node)

    order = []

    while available:
        node = _pop(available)
        order.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                _push(available, neighbor)

    if len(order) != len(indegree):
        raise ValueError("Cycle detected")

    return order
