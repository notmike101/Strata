```
def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in nodes}
    indegree = {node: 0 for node in nodes}

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("edges must contain (before, after) pairs")
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    available = [node for node in nodes if indegree[node] == 0]
    available.sort()

    def push(node):
        available.append(node)
        i = len(available) - 1
        while i > 0:
            p = (i - 1) // 2
            if available[i] < available[p]:
                available[i], available[p] = available[p], available[i]
                i = p
            else:
                break

    def pop():
        root = available[0]
        last = available.pop()
        if available:
            available[0] = last
            i = 0
            size = len(available)
            while True:
                left = 2 * i + 1
                right = left + 1
                smallest = i
                if left < size and available[left] < available[smallest]:
                    smallest = left
                if right < size and available[right] < available[smallest]:
                    smallest = right
                if smallest == i:
                    break
                available[i], available[smallest] = available[smallest], available[i]
                i = smallest
        return root

    order = []
    while available:
        node = pop()
        order.append(node)
        for successor in adjacency[node]:
            indegree[successor] -= 1
            if indegree[successor] == 0:
                push(successor)

    if len(order) != len(node_set):
        raise ValueError("cycle detected")
    return order
```
