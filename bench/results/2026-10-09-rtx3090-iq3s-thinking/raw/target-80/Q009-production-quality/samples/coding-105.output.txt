def topological_order(nodes, edges):
    node_list = list(nodes)
    node_set = set(node_list)
    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("Each edge must contain exactly two endpoints")
        if before not in node_set or after not in node_set:
            raise ValueError("Edge endpoint not in nodes")
        if after in adjacency[before]:
            continue
        adjacency[before].add(after)
        indegree[after] += 1

    heap = [node for node in node_set if indegree[node] == 0]
    heap.sort()

    def sift_down(i):
        n = len(heap)
        while True:
            left = 2 * i + 1
            if left >= n:
                break
            right = left + 1
            smallest = left
            if right < n and heap[right] < heap[left]:
                smallest = right
            if heap[i] <= heap[smallest]:
                break
            heap[i], heap[smallest] = heap[smallest], heap[i]
            i = smallest

    def sift_up(i):
        while i:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    order = []
    while heap:
        node = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            sift_down(0)
        order.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                heap.append(neighbor)
                sift_up(len(heap) - 1)

    if len(order) != len(node_set):
        raise ValueError("Cycle detected")
    return order
