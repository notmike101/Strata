def topological_order(nodes, edges):
    node_set = set(nodes)
    # validate edges and deduplicate
    adjacency = {node: [] for node in nodes}
    indegree = {node: 0 for node in nodes}
    seen_edges = set()
    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint not found")
        edge = (before, after)
        if edge in seen_edges:
            continue
        seen_edges.add(edge)
        adjacency[before].append(after)
        indegree[after] += 1

    total = len(node_set)
    heap = []

    def push(node):
        heap.append(node)
        i = len(heap) - 1
        while i:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    def pop():
        if not heap:
            return None
        result = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            i = 0
            while True:
                left = i * 2 + 1
                right = left + 1
                smallest = i
                if left < len(heap) and heap[left] < heap[smallest]:
                    smallest = left
                if right < len(heap) and heap[right] < heap[smallest]:
                    smallest = right
                if smallest == i:
                    break
                heap[i], heap[smallest] = heap[smallest], heap[i]
                i = smallest
        return result

    for node in nodes:
        if indegree[node] == 0:
            push(node)

    order = []
    while heap:
        node = pop()
        order.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                push(neighbor)

    if len(order) != total:
        raise ValueError("cycle detected")
    return order

```python
def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: [] for node in node_set}
    indegree = {node: 0 for node in node_set}
    seen_edges = set()

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint not in nodes")
        edge = (before, after)
        if edge in seen_edges:
            continue
        seen_edges.add(edge)
        adjacency[before].append(after)
        indegree[after] += 1

    heap = []

    def push(node):
        heap.append(node)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    def pop_node():
        if not heap:
            return None
        smallest = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            i = 0
            while True:
                left = i * 2 + 1
                right = left + 1
                smallest_idx = i
                if left < len(heap) and heap[left] < heap[smallest_idx]:
                    smallest_idx = left
                if right < len(heap) and heap[right] < heap[smallest_idx]:
                    smallest_idx = right
                if smallest_idx == i:
                    break
                heap[i], heap[smallest_idx] = heap[smallest_idx], heap[i]
                i = smallest_idx
        return smallest

    for node in node_set:
        if indegree[node] == 0:
            push(node)

    order = []
    while heap:
        node = pop_node()
        order.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                push(neighbor)

    if len(order) != len(node_set):
        raise ValueError("cycle detected")

    return order
```
