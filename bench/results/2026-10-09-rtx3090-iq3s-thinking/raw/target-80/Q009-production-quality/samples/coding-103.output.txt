def topological_order(nodes, edges):
    node_set = set(nodes)
    expected_count = len(node_set)

    indegree = {node: 0 for node in node_set}
    adjacency = {node: [] for node in node_set}
    seen_edges = set()

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("Edge must contain exactly two nodes") from None

        if before not in node_set or after not in node_set:
            raise ValueError("Edge endpoint absent from nodes")

        if before == after:
            raise ValueError("Cycle detected")

        edge_key = (before, after)
        if edge_key in seen_edges:
            continue
        seen_edges.add(edge_key)

        adjacency[before].append(after)
        indegree[after] += 1

    def _push(heap, item):
        heap.append(item)
        idx = len(heap) - 1
        while idx > 0:
            parent = (idx - 1) // 2
            if heap[idx] < heap[parent]:
                heap[idx], heap[parent] = heap[parent], heap[idx]
                idx = parent
            else:
                break

    def _pop(heap):
        item = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            idx = 0
            size = len(heap)
            while True:
                left = 2 * idx + 1
                right = left + 1
                smallest = idx
                if left < size and heap[left] < heap[smallest]:
                    smallest = left
                if right < size and heap[right] < heap[smallest]:
                    smallest = right
                if smallest == idx:
                    break
                heap[idx], heap[smallest] = heap[smallest], heap[idx]
                idx = smallest
        return item

    heap = []
    for node in node_set:
        if indegree[node] == 0:
            _push(heap, node)

    order = []
    while heap:
        node = _pop(heap)
        order.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                _push(heap, neighbor)

    if len(order) != expected_count:
        raise ValueError("Cycle detected")

    return order
