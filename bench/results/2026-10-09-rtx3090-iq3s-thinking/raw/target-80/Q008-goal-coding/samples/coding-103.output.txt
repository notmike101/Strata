def topological_order(nodes, edges):
    node_set = set(nodes)
    indegree = {node: 0 for node in nodes}
    outgoing = {node: set() for node in nodes}

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")
        if before == after:
            raise ValueError("cycle detected")
        if after not in outgoing[before]:
            outgoing[before].add(after)
            indegree[after] += 1

    def heap_push(heap, item):
        heap.append(item)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    def heap_pop(heap):
        root = heap[0]
        heap[0] = heap[-1]
        heap.pop()

        i = 0
        size = len(heap)
        while True:
            left = 2 * i + 1
            if left >= size:
                break

            right = left + 1
            smallest = left

            if right < size and heap[right] < heap[left]:
                smallest = right

            if heap[i] <= heap[smallest]:
                break

            heap[i], heap[smallest] = heap[smallest], heap[i]
            i = smallest

        return root

    heap = []
    for node in indegree:
        if indegree[node] == 0:
            heap_push(heap, node)

    order = []
    while heap:
        node = heap_pop(heap)
        order.append(node)

        for neighbor in outgoing[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                heap_push(heap, neighbor)

    if len(order) != len(node_set):
        raise ValueError("cycle detected")

    return order
