def topological_order(nodes, edges):
    node_set = set(nodes)

    adjacency = {node: [] for node in node_set}
    indegree = {node: 0 for node in node_set}
    seen_edges = set()

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("edges must be pairs") from None
        try:
            if before not in node_set or after not in node_set:
                raise ValueError("edge endpoint not in nodes")
        except TypeError:
            raise ValueError("edge endpoint not in nodes") from None
        if before == after:
            raise ValueError("cycle detected")

        edge_key = (before, after)
        if edge_key not in seen_edges:
            seen_edges.add(edge_key)
            adjacency[before].append(after)
            indegree[after] += 1

    def _sift_down(index, size):
        while True:
            left = 2 * index + 1
            right = left + 1
            smallest = index
            if left < size and available[left] < available[smallest]:
                smallest = left
            if right < size and available[right] < available[smallest]:
                smallest = right
            if smallest == index:
                return
            available[index], available[smallest] = available[smallest], available[index]
            index = smallest

    def _sift_up(index):
        while index > 0:
            parent = (index - 1) // 2
            if available[index] >= available[parent]:
                return
            available[index], available[parent] = available[parent], available[index]
            index = parent

    available = [node for node in node_set if indegree[node] == 0]
    for i in range(len(available) // 2 - 1, -1, -1):
         _sift_down(i, len(available))

    result=[]
    while available:
        node = available[0]
        last = available.pop()
        if available:
            available[0] = last
            _sift_down(0, len(available))
        result.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                available.append(neighbor)
                _sift_up(len(available)-1)

    if len(result) != len(node_set):
        raise ValueError("cycle detected")
    return result
