def topological_order(nodes, edges):
    node_set = set(nodes)

    indegree = {node: 0 for node in node_set}
    adjacency = {node: set() for node in node_set}

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("each edge must contain exactly two endpoints")

        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint not found in nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def _sift_down(heap, index):
        size = len(heap)
        while True:
            left = index * 2 + 1
            right = left + 1
            smallest = index

            if left < size and heap[left] < heap[smallest]:
                smallest = left
            if right < size and heap[right] < heap[smallest]:
                smallest = right

            if smallest == index:
                return

            heap[index], heap[smallest] = heap[smallest], heap[index]
            index = smallest

    def _sift_up(heap, index):
        while index > 0:
            parent = (index - 1) // 2
            if heap[parent] <= heap[index]:
                return
            heap[parent], heap[index] = heap[index], heap[parent]
            index = parent

    heap = sorted(node_set)
    order = []

    while heap:
        node = heap[0]
        heap[0] = heap[-1]
        heap.pop()
        if heap:
            _sift_down(heap, 0)

        order.append(node)

        for successor in adjacency[node]:
            indegree[successor] -= 1
            if indegree[successor] == 0:
                heap.append(successor)
                _sift_up(heap, len(heap) - 1)

    if len(order) != len(node_set):
        raise ValueError("cycle detected")

    return order
