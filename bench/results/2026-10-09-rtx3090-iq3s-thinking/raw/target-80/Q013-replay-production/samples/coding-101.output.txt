def topological_order(nodes, edges):
    nodes_set = set(nodes)
    indegree = {node: 0 for node in nodes_set}
    adjacency = {node: set() for node in nodes_set}

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("edges must contain pairs") from None

        try:
            valid_endpoint = before in nodes_set and after in nodes_set
        except TypeError:
            valid_endpoint = False

        if not valid_endpoint:
            raise ValueError("edge endpoint absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    heap = [node for node in nodes_set if indegree[node] == 0]
    heap.sort()

    def push_heap(node):
        heap.append(node)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= node:
                break
            heap[i] = heap[parent]
            i = parent
        heap[i] = node

    def pop_heap():
        root = heap[0]
        last = heap.pop()
        if not heap:
            return root

        heap[0] = last
        i = 0
        size = len(heap)

        while True:
            left = 2 * i + 1
            right = left + 1
            smallest = i

            if left < size and heap[left] < heap[smallest]:
                smallest = left
            if right < size and heap[right] < heap[smallest]:
                smallest = right

            if smallest == i:
                break

            heap[i], heap[smallest] = heap[smallest], heap[i]
            i = smallest

        return root

    order = []

    while heap:
        node = pop_heap()
        order.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                push_heap(neighbor)

    if len(order) != len(nodes_set):
        raise ValueError("cycle detected")

    return order
