def topological_order(nodes, edges):
    unique_nodes = []
    node_set = set()
    for node in nodes:
        if node not in node_set:
            node_set.add(node)
            unique_nodes.append(node)

    indegree = {node: 0 for node in unique_nodes}
    adjacency = {node: set() for node in unique_nodes}

    for edge in edges:
        before, after = edge
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def push(heap, value):
        heap.append(value)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[i] < heap[parent]:
                heap[i], heap[parent] = heap[parent], heap[i]
                i = parent
            else:
                break

    def pop(heap):
        root = heap[0]
        heap[0] = heap[-1]
        heap.pop()
        if heap:
            i = 0
            size = len(heap)
            while True:
                left = 2 * i + 1
                right = left + 1
                smallest = i
                if left < size and heap[left] < heap[smallest]:
                    smallest = left
                if right < size and heap[right] < heap[smallest]:
                    smallest = right
                if smallest == i:
                    break
                heap[i], heap[smallest] = heap[smallest], heap[i]
                i = smallest
        return root

    heap = []
    for node in unique_nodes:
        if indegree[node] == 0:
            push(heap, node)

    order = []
    while heap:
        node = pop(heap)
        order.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                push(heap, neighbor)

    if len(order) != len(unique_nodes):
        raise ValueError("cycle detected")
    return order

def topological_order(nodes, edges):
    unique_nodes = []
    node_set = set()
    for node in nodes:
        if node not in node_set:
            node_set.add(node)
            unique_nodes.append(node)

    indegree = {node: 0 for node in unique_nodes}
    adjacency = {node: set() for node in unique_nodes}

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def _heap_push(heap, value):
        heap.append(value)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[i] < heap[parent]:
                heap[i], heap[parent] = heap[parent], heap[i]
                i = parent
            else:
                break

    def _heap_pop(heap):
        root = heap[0]
        heap[0] = heap[-1]
        heap.pop()

        if heap:
            i = 0
            size = len(heap)
            while True:
                left = 2 * i + 1
                right = left + 1
                smallest = i

                if left < size and heap[left] < heap[smallest]:
                    smallest = left
                if right < size and heap[right] < heap[smallest]:
                    smallest = right

                if smallest == i:
                    break

                heap[i], heap[smallest] = heap[smallest], heap[i]
                i = smallest

        return root

    heap = []
    for node in unique_nodes:
        if indegree[node] == 0:
            _heap_push(heap, node)

    order = []
    while heap:
        node = _heap_pop(heap)
        order.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                _heap_push(heap, neighbor)

    if len(order) != len(unique_nodes):
        raise ValueError("cycle detected")

    return order
