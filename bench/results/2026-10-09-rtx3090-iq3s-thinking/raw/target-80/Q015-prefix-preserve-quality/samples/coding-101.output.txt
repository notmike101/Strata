        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

        i = start
        while True:
            left = 2 * i + 1

            right = left + 1

            if right < size and heap[right] < heap[left]:

            if heap[i] <= heap[smallest]:
                break

            i = smallest

def topological_order(nodes, edges):
    node_set = set(nodes)
    if len(node_set) != len(nodes):
        raise ValueError("nodes must be unique")

    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    heap = [node for node in node_set if indegree[node] == 0]

    def sift_down(i):
        size = len(heap)
        while True:
            left = 2 * i + 1
            if left >= size:
                break
            right = left + 1
            child = left
            if right < size and heap[right] < heap[left]:
                child = right
            if heap[i] <= heap[child]:
                break
            heap[i], heap[child] = heap[child], heap[i]
            i = child

    for i in range(len(heap) // 2 - 1, -1, -1):
        sift_down(i)

    def push(item):
        heap.append(item)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    def pop():
        item = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            sift_down(0)
        return item

    order = []
    while heap:
        node = pop()
        order.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                push(neighbor)

    if len(order) != len(node_set):
        raise ValueError("cycle detected")
    return order

    node_set = set(nodes)
    indegree = {node: 0 for node in node_set}
    adjacency = {}

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")
        if before not in adjacency:
            adjacency[before] = set()
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    heap = [node for node in node_set if indegree[node] == 0]

    def sift_down(i):
        size = len(heap)
        while True:
            left = 2 * i + 1
            if left >= size:
                break
            right = left + 1
            child = left
            if right < size and heap[right] < heap[left]:
                child = right
            if heap[i] <= heap[child]:
                break
            heap[i], heap[child] = heap[child], heap[i]
            i = child

    for i in range(len(heap) // 2 - 1, -1, -1):
        sift_down(i)

    def push(item):
        heap.append(item)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    def pop():

        last = heap.pop()
        if heap:
            heap[0] = last
            sift_down(0)
        return item

        order.append(node)
        for neighbor in adjacency.get(node, ()):
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:

def topological_order(nodes, edges):
    node_set = set(nodes)
    indegree = {node: 0 for node in node_set}
    adjacency = {}

    for before, after in edges:
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")

        if before not in adjacency:
            adjacency[before] = set()

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    if not adjacency:
        return sorted(node_set)

    heap = sorted(node for node in node_set if indegree[node] == 0)

    def sift_down(i):
        size = len(heap)
        while True:
            left = 2 * i + 1
            if left >= size:
                break

            right = left + 1
            child = left
            if right < size and heap[right] < heap[left]:
                child = right

            if heap[i] <= heap[child]:
                break

            heap[i], heap[child] = heap[child], heap[i]
            i = child

    def push(item):
        heap.append(item)
        i = len(heap) - 1
        while i > 0:
            parent = (i - 1) // 2
            if heap[parent] <= heap[i]:
                break
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent

    def pop():
        item = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            sift_down(0)
        return item

    order = []

    while heap:
        node = pop()
        order.append(node)

        for neighbor in adjacency.get(node, ()):
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                push(neighbor)

    if len(order) != len(node_set):
        raise ValueError("cycle detected")

    return order
