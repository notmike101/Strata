```python
def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for edge in edges:
        try:
            before, after = edge
        except (TypeError, ValueError):
            raise ValueError("each edge must be a (before, after) pair")
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def heap_push(heap, value):
        heap.append(value)
        index = len(heap) - 1
        while index > 0:
            parent = (index - 1) // 2
            if heap[index] >= heap[parent]:
                break
            heap[index], heap[parent] = heap[parent], heap[index]
            index = parent

    def heap_pop(heap):
        root = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            index = 0
            size = len(heap)
            while True:
                left = 2 * index + 1
                right = 2 * index + 2
                smallest = index
                if left < size and heap[left] < heap[smallest]:
                    smallest = left
                if right < size and heap[right] < heap[smallest]:
                    smallest = right
                if smallest == index:
                    break
                heap[index], heap[smallest] = heap[smallest], heap[index]
                index = smallest
        return root

    available = []
    for node in nodes:
        if indegree[node] == 0:
            heap_push(available, node)

    ordered = []
    while available:
        node = heap_pop(available)
        ordered.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                heap_push(available, neighbor)

    if len(ordered) != len(node_set):
        raise ValueError("cycle detected")
    return ordered
```

```python
def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for edge in edges:
        if not isinstance(edge, (tuple, list)) or len(edge) != 2:
            raise ValueError("edges must be (before, after) pairs")
        before, after = edge
        if before not in node_set or after not in node_set:
            raise ValueError("edge endpoint absent from nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def push(heap, value):
        heap.append(value)
        i = len(heap) - 1
        while i:
            p = (i - 1) // 2
            if heap[i] >= heap[p]:
                break
            heap[i], heap[p] = heap[p], heap[i]
            i = p

    def pop(heap):
        root = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            i = 0
            n = len(heap)
            while True:
                left = 2 * i + 1
                right = left + 1
                smallest = i
                if left < n and heap[left] < heap[smallest]:

                if right < n and heap[right] < heap[smallest]:
                    smallest = right
                if smallest == i:
                    break

    if len(order) != len(node_set):
```

```python
def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for edge in edges:
        if not isinstance(edge, (tuple, list)) or len(edge) != 2:
            raise ValueError("each edge must be a (before, after) pair")

        before, after = edge
        try:
            before_ok = before in node_set
            after_ok = after in node_set
        except TypeError:
            raise ValueError("edge endpoint absent from nodes") from None

        if not before_ok or not after_ok:
            raise ValueError("edge endpoint absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def push(heap, value):
        heap.append(value)
        index = len(heap) - 1
        while index > 0:
            parent = (index - 1) // 2
            if heap[index] >= heap[parent]:
                break
            heap[index], heap[parent] = heap[parent], heap[index]
            index = parent

    def pop(heap):
        root = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            index = 0
            size = len(heap)
            while True:
                left = 2 * index + 1
                right = left + 1
                smallest = index
                if left < size and heap[left] < heap[smallest]:
                    smallest = left
                if right < size and heap[right] < heap[smallest]:
                    smallest = right
                if smallest == index:
                    break
                heap[index], heap[smallest] = heap[smallest], heap[index]
                index = smallest
        return root

    available = []
    for node in node_set:
        if indegree[node] == 0:
            push(available, node)

    ordered = []
    while available:
        node = pop(available)
        ordered.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                push(available, neighbor)

    if len(ordered) != len(node_set):
        raise ValueError("cycle detected")

    return ordered
```

```python
def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for edge in edges:
        if not isinstance(edge, (tuple, list)) or len(edge) != 2:
            raise ValueError("each edge must be a (before, after) pair")
        before, after = edge
        try:
            before_ok = before in node_set
            after_ok = after in node_set
        except TypeError:
            raise ValueError("edge endpoint absent from nodes") from None
        if not before_ok or not after_ok:
            raise ValueError("edge endpoint absent from nodes")
        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def sift_down(heap, index, size):
        while True:
            left = 2 * index + 1
            right = left + 1
            smallest = index
            if left < size and heap[left] < heap[smallest]:
                smallest = left
            if right < size and heap[right] < heap[smallest]:
                smallest = right
            if smallest == index:
                break
            heap[index], heap[smallest] = heap[smallest], heap[index]
            index = smallest

    def heap_push(heap, value):
        heap.append(value)
        index = len(heap) - 1
        while index > 0:
            parent = (index - 1) // 2
            if heap[index] >= heap[parent]:
                break
            heap[index], heap[parent] = heap[parent], heap[index]
            index = parent

    def heap_pop(heap):
        root = heap[0]
        last = heap.pop()
        if heap:
            heap[0] = last
            sift_down(heap, 0, len(heap))
        return root

    available = [node for node in node_set if indegree[node] == 0]
    for index in range(len(available) // 2 - 1, -1, -1):
        sift_down(available, index, len(available))

    ordered = []
    while available:
        node = heap_pop(available)
        ordered.append(node)
        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                heap_push(available, neighbor)

    if len(ordered) != len(node_set):
        raise ValueError("cycle detected")
    return ordered
```

```python
def topological_order(nodes, edges):
    node_set = set(nodes)
    adjacency = {node: set() for node in node_set}
    indegree = {node: 0 for node in node_set}

    for edge in edges:
        if not isinstance(edge, (tuple, list)) or len(edge) != 2:
            raise ValueError("each edge must be a (before, after) pair")

        before, after = edge

        try:
            before_ok = before in node_set
            after_ok = after in node_set
        except TypeError:
            raise ValueError("edge endpoint absent from nodes") from None

        if not before_ok or not after_ok:
            raise ValueError("edge endpoint absent from nodes")

        if after not in adjacency[before]:
            adjacency[before].add(after)
            indegree[after] += 1

    def sift_down(heap, index, size):
        while True:
            left = 2 * index + 1
            right = left + 1
            smallest = index

            if left < size and heap[left] < heap[smallest]:
                smallest = left
            if right < size and heap[right] < heap[smallest]:
                smallest = right

            if smallest == index:
                break

            heap[index], heap[smallest] = heap[smallest], heap[index]
            index = smallest

    def heap_push(heap, value):
        heap.append(value)
        index = len(heap) - 1

        while index > 0:
            parent = (index - 1) // 2
            if heap[index] >= heap[parent]:
                break
            heap[index], heap[parent] = heap[parent], heap[index]
            index = parent

    def heap_pop(heap):
        root = heap[0]
        last = heap.pop()

        if heap:
            heap[0] = last
            sift_down(heap, 0, len(heap))

        return root

    available = [node for node in node_set if indegree[node] == 0]

    for index in range(len(available) // 2 - 1, -1, -1):
        sift_down(available, index, len(available))

    ordered = []

    while available:
        node = heap_pop(available)
        ordered.append(node)

        for neighbor in adjacency[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                heap_push(available, neighbor)

    if len(ordered) != len(node_set):
        raise ValueError("cycle detected")

    return ordered
```
